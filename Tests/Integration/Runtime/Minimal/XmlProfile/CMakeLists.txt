cmake_minimum_required(VERSION 3.24)
project(LVGLXmlProfile LANGUAGES C CXX)
if(NOT IS_DIRECTORY "${LVGL_PB_COMPONENT_DIR}")
    message(FATAL_ERROR "Pass LVGL_PB_COMPONENT_DIR for the generated label profile")
endif()
add_subdirectory("${LVGL_PB_COMPONENT_DIR}" component)
add_executable(LVGLXmlProfileConsumer Consumer.cpp)
target_link_libraries(LVGLXmlProfileConsumer PRIVATE LVGLProtobuf::Embedded)
target_compile_features(LVGLXmlProfileConsumer PRIVATE cxx_std_17)
if(MSVC)
    target_compile_options(LVGLXmlProfileConsumer PRIVATE /utf-8 /Zc:preprocessor /Zc:__cplusplus)
endif()
enable_testing()
add_test(NAME LVGLXmlProfileConsumer COMMAND LVGLXmlProfileConsumer "${XML_FIXTURE}")
