cmake_minimum_required(VERSION 3.24)
project(LVGLMinimalRuntimeConsumer LANGUAGES C CXX)
if(NOT IS_DIRECTORY "${LVGL_PB_COMPONENT_DIR}")
    message(FATAL_ERROR "Pass LVGL_PB_COMPONENT_DIR for a generated minimal component")
endif()
add_subdirectory("${LVGL_PB_COMPONENT_DIR}" component)
option(LVGL_PB_CODES_TEST "Exercise a barcode/QR profile instead of a label-only profile" OFF)
option(LVGL_PB_ADVANCED_KEYBOARD_TEST "Exercise the portable Advanced Keyboard profile" OFF)
if(LVGL_PB_CODES_TEST AND LVGL_PB_ADVANCED_KEYBOARD_TEST)
    message(FATAL_ERROR "Select one minimal component test profile")
endif()
if(LVGL_PB_ADVANCED_KEYBOARD_TEST)
    add_executable(LVGLMinimalRuntimeConsumer KeyboardConsumer.cpp)
elseif(LVGL_PB_CODES_TEST)
    add_executable(LVGLMinimalRuntimeConsumer CodeConsumer.cpp)
else()
    add_executable(LVGLMinimalRuntimeConsumer Consumer.cpp)
endif()
target_link_libraries(LVGLMinimalRuntimeConsumer PRIVATE LVGLProtobuf::Embedded)
target_compile_features(LVGLMinimalRuntimeConsumer PRIVATE cxx_std_17)
if(MSVC)
    target_compile_options(LVGLMinimalRuntimeConsumer PRIVATE /utf-8 /Zc:preprocessor /Zc:__cplusplus)
    target_link_options(LVGLMinimalRuntimeConsumer PRIVATE /DEBUG:FULL "/MAP:${CMAKE_CURRENT_BINARY_DIR}/consumer.map")
endif()
enable_testing()
add_test(NAME LVGLMinimalRuntimeConsumer COMMAND LVGLMinimalRuntimeConsumer)
