cmake_minimum_required(VERSION 3.24)
project(RuntimeVersionTests LANGUAGES C CXX)
add_library(lvgl STATIC IMPORTED GLOBAL)
set_target_properties(lvgl PROPERTIES IMPORTED_LOCATION "${LVGL_ARCHIVE}" INTERFACE_INCLUDE_DIRECTORIES "${LVGL_SOURCE_DIR}"
    INTERFACE_COMPILE_DEFINITIONS "LV_CONF_SKIP=1;LV_COLOR_DEPTH=32;LV_COLOR_FORMAT_DEFAULT=LV_COLOR_FORMAT_ARGB8888;LV_USE_STDLIB_MALLOC=1;LV_MEM_CUSTOM=1;LV_GRADIENT_MAX_STOPS=8;LV_USE_CALENDAR_HEADER_ARROW=1;LV_USE_RLE=1;LV_USE_LZ4=1;LV_USE_LZ4_INTERNAL=1;LV_BIN_DECODER_RAM_LOAD=1;LV_USE_LODEPNG=1;LV_USE_TJPGD=1")
set(BUILD_TESTING OFF)
add_subdirectory("${REPOSITORY_ROOT}/Runtime" runtime)
add_library(Catch STATIC IMPORTED)
set_target_properties(Catch PROPERTIES IMPORTED_LOCATION "${CATCH_ARCHIVE}" INTERFACE_INCLUDE_DIRECTORIES "${REPOSITORY_ROOT}/3rdParty/Catch2/single_include/catch2")
add_executable(unit "${REPOSITORY_ROOT}/Tests/UnitTests/Runtime/Main.cpp" "${REPOSITORY_ROOT}/Tests/UnitTests/Runtime/API.cpp")
target_link_libraries(unit PRIVATE LVGLRuntime::Runtime LVSerial Catch)
target_include_directories(unit PRIVATE "${REPOSITORY_ROOT}/Runtime/compat")
add_executable(functional Functional.cpp)
target_include_directories(functional PRIVATE "${REPOSITORY_ROOT}/Runtime/compat")
target_link_libraries(functional PRIVATE LVGLRuntime::Runtime LVSerial)
if(UNIX)
    target_link_libraries(lvgl INTERFACE m)
endif()
enable_testing()
add_test(NAME Unit COMMAND unit "[runtime]")
add_test(NAME Widgets COMMAND functional "${WIDGET_PROJECT}" widgets)
add_test(NAME Assets COMMAND functional "${ASSET_PROJECT}" assets)
set_tests_properties(Unit Widgets Assets PROPERTIES TIMEOUT 60)
add_executable(c_firmware "${REPOSITORY_ROOT}/Tests/Integration/Runtime/Standalone/Main.c")
target_link_libraries(c_firmware PRIVATE LVGLRuntime::Runtime)
add_test(NAME CPublicAPI COMMAND c_firmware "${ASSET_PROJECT}")
set_tests_properties(CPublicAPI PROPERTIES TIMEOUT 30)
if(MSVC)
    foreach(test unit functional)
        target_compile_options(${test} PRIVATE /Zi)
        target_link_options(${test} PRIVATE /DEBUG:FULL)
    endforeach()
endif()
add_executable(media MediaReplay.cpp)
target_include_directories(media PRIVATE "${REPOSITORY_ROOT}/Runtime/compat")
target_link_libraries(media PRIVATE LVGLRuntime::Runtime LVSerial)
if(MSVC)
    target_compile_options(media PRIVATE /Zi)
    target_link_options(media PRIVATE /DEBUG:FULL)
endif()
add_test(NAME Media COMMAND media "${MEDIA_CASES}")
set_tests_properties(Media PROPERTIES TIMEOUT 60)
