cmake_minimum_required(VERSION 3.24)
project(GeneratedMemoryTests LANGUAGES C)
option(UI_MEMORY_EXPECT_OPTIMIZED "Require optimized immutable asset storage" ON)
option(UI_MEMORY_EXPECT_STATIC_TEXT "Require immutable text instead of editable text" ${UI_MEMORY_EXPECT_OPTIMIZED})
foreach(required LVGL_ARCHIVE LVGL_SOURCE_DIR GENERATED_DIR)
    if(NOT DEFINED ${required} OR "${${required}}" STREQUAL "")
        message(FATAL_ERROR "Set ${required} before configuring generated memory tests")
    endif()
endforeach()
add_library(lvgl STATIC IMPORTED GLOBAL)
set_target_properties(lvgl PROPERTIES IMPORTED_LOCATION "${LVGL_ARCHIVE}"
    INTERFACE_INCLUDE_DIRECTORIES "${LVGL_SOURCE_DIR}"
    INTERFACE_COMPILE_DEFINITIONS "LV_CONF_SKIP=1;LV_COLOR_DEPTH=32;LV_USE_STDLIB_MALLOC=1;LV_MEM_CUSTOM=1;LV_GRADIENT_MAX_STOPS=8;LV_USE_CALENDAR_HEADER_ARROW=1;LV_USE_RLE=1;LV_USE_LZ4=1;LV_USE_LZ4_INTERNAL=1;LV_BIN_DECODER_RAM_LOAD=1;LV_USE_LODEPNG=1;LV_USE_TJPGD=1")
if(EXISTS "${LVGL_SOURCE_DIR}/include/lvgl/config/lv_conf_internal.h")
    target_compile_definitions(lvgl INTERFACE LV_COLOR_FORMAT_DEFAULT=LV_COLOR_FORMAT_ARGB8888)
endif()
if(UNIX)
    target_link_libraries(lvgl INTERFACE m)
endif()
foreach(memory_target generated_memory generated_memory_mutable)
    add_executable(${memory_target} main.c "${GENERATED_DIR}/ui.c" "${GENERATED_DIR}/ui_assets.c")
    target_include_directories(${memory_target} PRIVATE "${GENERATED_DIR}")
    target_link_libraries(${memory_target} PRIVATE lvgl)
    target_compile_definitions(${memory_target} PRIVATE "UI_MEMORY_EXPECT_OPTIMIZED=$<BOOL:${UI_MEMORY_EXPECT_OPTIMIZED}>")
    if(memory_target STREQUAL "generated_memory_mutable")
        target_compile_definitions(${memory_target} PRIVATE UI_USE_STATIC_TEXT=0 UI_MEMORY_EXPECT_STATIC_TEXT=0)
    else()
        target_compile_definitions(${memory_target} PRIVATE "UI_MEMORY_EXPECT_STATIC_TEXT=$<BOOL:${UI_MEMORY_EXPECT_STATIC_TEXT}>")
    endif()
    if(EXISTS "${LVGL_SOURCE_DIR}/src/widgets/label/lv_label_private.h")
        target_compile_definitions(${memory_target} PRIVATE UI_MEMORY_PRIVATE_WIDGETS=1)
    endif()
    set_target_properties(${memory_target} PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES)
    if(MSVC)
        target_compile_options(${memory_target} PRIVATE /W3 /we4013 /utf-8)
    endif()
endforeach()
enable_testing()
add_test(NAME GeneratedMemory COMMAND generated_memory)
add_test(NAME GeneratedMemoryMutableText COMMAND generated_memory_mutable)
set_tests_properties(GeneratedMemory GeneratedMemoryMutableText PROPERTIES TIMEOUT 30)
