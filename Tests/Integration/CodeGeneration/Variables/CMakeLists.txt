cmake_minimum_required(VERSION 3.24)
project(GeneratedVariableTests LANGUAGES C CXX)
foreach(required LVGL_ARCHIVE LVGL_SOURCE_DIR GENERATED_DIR)
    if(NOT DEFINED ${required} OR "${${required}}" STREQUAL "")
        message(FATAL_ERROR "Set ${required} before configuring generated variable tests")
    endif()
endforeach()
add_library(lvgl STATIC IMPORTED GLOBAL)
set_target_properties(lvgl PROPERTIES IMPORTED_LOCATION "${LVGL_ARCHIVE}"
    INTERFACE_INCLUDE_DIRECTORIES "${LVGL_SOURCE_DIR}"
    INTERFACE_COMPILE_DEFINITIONS "LV_CONF_SKIP=1;LV_COLOR_DEPTH=32;LV_USE_STDLIB_MALLOC=1;LV_MEM_CUSTOM=1;LV_GRADIENT_MAX_STOPS=8;LV_USE_CALENDAR_HEADER_ARROW=1;LV_USE_RLE=1;LV_USE_LZ4=1;LV_USE_LZ4_INTERNAL=1;LV_BIN_DECODER_RAM_LOAD=1;LV_USE_LODEPNG=1;LV_USE_TJPGD=1")
if(EXISTS "${LVGL_SOURCE_DIR}/include/lvgl/config/lv_conf_internal.h")
    target_compile_definitions(lvgl INTERFACE LV_COLOR_FORMAT_DEFAULT=LV_COLOR_FORMAT_ARGB8888)
endif()
if(UNIX)
    target_link_libraries(lvgl INTERFACE m)
endif()
add_executable(generated_variables main.c CppPublic.cpp GeneratedWrapper.c "${GENERATED_DIR}/ui_assets.c")
target_link_libraries(generated_variables PRIVATE lvgl)
target_include_directories(generated_variables PRIVATE "${GENERATED_DIR}")
set_target_properties(generated_variables PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO
    CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
if(MSVC)
    target_compile_options(generated_variables PRIVATE /W3 /we4013 /utf-8)
else()
    target_compile_options(generated_variables PRIVATE -Wall -Wextra $<$<COMPILE_LANGUAGE:C>:-Werror=implicit-function-declaration>)
endif()
enable_testing()
add_test(NAME GeneratedVariables COMMAND generated_variables)
set_tests_properties(GeneratedVariables PROPERTIES TIMEOUT 30)
