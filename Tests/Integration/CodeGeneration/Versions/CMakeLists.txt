cmake_minimum_required(VERSION 3.24)
project(LVGLVersionGeneratorTests LANGUAGES C CXX)
file(GLOB_RECURSE lvgl_sources "${LVGL_SOURCE_DIR}/src/*.c")
add_library(lvgl STATIC ${lvgl_sources})
target_include_directories(lvgl PUBLIC "${LVGL_SOURCE_DIR}")
target_compile_definitions(lvgl PUBLIC LV_CONF_SKIP=1 LV_COLOR_DEPTH=32 LV_USE_STDLIB_MALLOC=1 LV_MEM_CUSTOM=1 LV_GRADIENT_MAX_STOPS=8 LV_USE_CALENDAR_HEADER_ARROW=1 LV_USE_RLE=1 LV_USE_LZ4=1 LV_USE_LZ4_INTERNAL=1 LV_BIN_DECODER_RAM_LOAD=1 LV_USE_LODEPNG=1 LV_USE_TJPGD=1)
if(EXISTS "${LVGL_SOURCE_DIR}/include/lvgl/config/lv_conf_internal.h")
    target_compile_definitions(lvgl PUBLIC LV_COLOR_FORMAT_DEFAULT=LV_COLOR_FORMAT_ARGB8888)
endif()
if(MSVC)
    target_compile_options(lvgl PRIVATE /W0 /utf-8 /Zi)
endif()
foreach(kind main assets)
    if(kind STREQUAL main)
        set(input "${GENERATED_DIR}")
    else()
        set(input "${ASSET_DIR}")
    endif()
    add_executable(test_${kind} Versioned.c "${input}/ui.c" "${input}/ui_assets.c")
    target_link_libraries(test_${kind} PRIVATE lvgl)
    target_include_directories(test_${kind} PRIVATE "${input}" "${FIXTURE_DIR}")
    target_compile_definitions(test_${kind} PRIVATE "TEST_${kind}=1")
    add_executable(test_cpp_${kind} CppPublic.cpp "${input}/ui.c" "${input}/ui_assets.c")
    target_link_libraries(test_cpp_${kind} PRIVATE lvgl)
    target_include_directories(test_cpp_${kind} PRIVATE "${input}")
    target_compile_definitions(test_cpp_${kind} PRIVATE "TEST_${kind}=1")
    target_compile_features(test_cpp_${kind} PRIVATE cxx_std_17)
    set_target_properties(test_cpp_${kind} PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
    set_target_properties(test_${kind} PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
    if(MSVC)
        target_compile_options(test_${kind} PRIVATE /W3 /we4013 /utf-8 /Zi
        )
        target_link_options(test_${kind} PRIVATE /DEBUG:FULL)
    endif()
endforeach()
if(UNIX)
    target_link_libraries(lvgl PUBLIC m)
endif()
enable_testing()
add_test(NAME GeneratedWidgets COMMAND test_main)
add_test(NAME GeneratedAssets COMMAND test_assets)
add_test(NAME CppWidgets COMMAND test_cpp_main)
add_test(NAME CppAssets COMMAND test_cpp_assets)
set_tests_properties(GeneratedWidgets GeneratedAssets CppWidgets CppAssets PROPERTIES TIMEOUT 30)
