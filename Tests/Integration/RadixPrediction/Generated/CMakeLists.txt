cmake_minimum_required(VERSION 3.24)
project(RadixGeneratedConsumer LANGUAGES C CXX)
add_library(lvgl STATIC IMPORTED GLOBAL)
set_target_properties(lvgl PROPERTIES IMPORTED_LOCATION "${LVGL_ARCHIVE}")
target_include_directories(lvgl INTERFACE "${LVGL_SOURCE_DIR}")
target_compile_definitions(lvgl INTERFACE LV_CONF_SKIP=1 LV_COLOR_DEPTH=32 LV_USE_STDLIB_MALLOC=1 LV_MEM_CUSTOM=1
    LV_GRADIENT_MAX_STOPS=8 LV_USE_CALENDAR_HEADER_ARROW=1 LV_USE_RLE=1 LV_USE_LZ4=1 LV_USE_LZ4_INTERNAL=1
    LV_BIN_DECODER_RAM_LOAD=1 LV_USE_LODEPNG=1 LV_USE_TJPGD=1 LV_COLOR_FORMAT_DEFAULT=LV_COLOR_FORMAT_ARGB8888)
if(UNIX)
    target_link_libraries(lvgl INTERFACE m)
endif()
add_subdirectory("${EXPORTED_DIR}" exported)
add_executable(radix_consumer ../Consumer.c HeaderConsumer.cpp)
target_compile_features(radix_consumer PRIVATE cxx_std_17)
set_target_properties(radix_consumer PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES)
target_link_libraries(radix_consumer PRIVATE generated_ui)
get_filename_component(profile "${EXPORTED_DIR}" NAME)
if(profile STREQUAL "fixed-v2")
    target_compile_definitions(radix_consumer PRIVATE RADIX_EXPECT_MASS=19u)
elseif(profile STREQUAL "radix-v4")
    target_compile_definitions(radix_consumer PRIVATE RADIX_EXPECT_MASS=4294967295u)
else()
    target_compile_definitions(radix_consumer PRIVATE RADIX_EXPECT_MASS=0u)
endif()
if(MSVC)
    target_compile_options(radix_consumer PRIVATE /utf-8 /we4013)
endif()
enable_testing()
add_test(NAME RadixGeneratedCConsumer COMMAND radix_consumer)
