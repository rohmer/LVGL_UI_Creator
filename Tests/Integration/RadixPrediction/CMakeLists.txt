cmake_minimum_required(VERSION 3.24)
project(RadixPredictionTests LANGUAGES CXX)
get_filename_component(root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${root}/cmake/AdvancedKeyboardDependency.cmake")
set(core "${ADVKBD_SOURCE_ROOT}/src")
set(codec "${ADVKBD_SOURCE_ROOT}/tools/ModelCodec")
set(fixtures "${ADVKBD_SOURCE_ROOT}/tests/native/radix")
add_executable(radix_prediction "${fixtures}/Main.cpp" "${core}/PredictionModel.cpp" "${core}/PredictionPrefix.cpp"
    "${core}/Utf8Fold.cpp" "${codec}/RadixModelBuilder.cpp")
add_executable(radix_cursor_cache "${fixtures}/CursorCache.cpp" "${core}/PredictionModel.cpp"
    "${core}/PredictionPrefix.cpp" "${codec}/RadixModelBuilder.cpp")
target_compile_features(radix_prediction PRIVATE cxx_std_17)
target_compile_features(radix_cursor_cache PRIVATE cxx_std_17)
target_compile_definitions(radix_prediction PRIVATE ESPOS_KEYBOARD_HEADLESS_PREDICTION=1
    LVSERIAL_USE_ADVANCED_KEYBOARD_PREDICTION=1 LVSERIAL_USE_ADVANCED_KEYBOARD_BUILTIN_TRIE=0)
target_compile_definitions(radix_cursor_cache PRIVATE ESPOS_KEYBOARD_HEADLESS_PREDICTION=1
    LVSERIAL_USE_ADVANCED_KEYBOARD_PREDICTION=1 LVSERIAL_USE_ADVANCED_KEYBOARD_BUILTIN_TRIE=0)
target_include_directories(radix_prediction PRIVATE "${core}" "${ADVKBD_SOURCE_ROOT}/include" "${codec}")
target_include_directories(radix_cursor_cache PRIVATE "${core}" "${ADVKBD_SOURCE_ROOT}/include" "${codec}")
if(MSVC)
    target_compile_options(radix_prediction PRIVATE /utf-8 /W4)
    target_compile_options(radix_cursor_cache PRIVATE /utf-8 /W4)
endif()
enable_testing()
add_test(NAME RadixPrediction COMMAND radix_prediction)
add_test(NAME RadixCursorCache COMMAND radix_cursor_cache)
set_tests_properties(RadixCursorCache PROPERTIES TIMEOUT 60 LABELS "prediction;radix;cache")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME RadixPredictionIndexParity COMMAND "${Python3_EXECUTABLE}" "${fixtures}/VerifyIndex.py" "$<TARGET_FILE:radix_prediction>")
