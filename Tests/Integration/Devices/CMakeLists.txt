cmake_minimum_required(VERSION 3.24)
project(DeviceToolsChecks LANGUAGES CXX)
enable_testing()
set(CMAKE_CXX_STANDARD 17)
find_package(Threads REQUIRED)
add_executable(DeviceToolsTests DeviceToolsTests.cpp ../../../UICreator/Devices/DeviceTools.cpp)
target_link_libraries(DeviceToolsTests PRIVATE Threads::Threads)
if(MSVC)
    target_compile_options(DeviceToolsTests PRIVATE /utf-8)
endif()
add_test(NAME DeviceToolsNativeProcesses COMMAND DeviceToolsTests "${CMAKE_CURRENT_SOURCE_DIR}/../../..")
set_tests_properties(DeviceToolsNativeProcesses PROPERTIES TIMEOUT 45)

add_executable(DeviceWorkflowTests Workflow.cpp ../../../UICreator/Devices/DeviceTargets.cpp ../../../UICreator/Devices/DeviceTools.cpp)
target_link_libraries(DeviceWorkflowTests PRIVATE Threads::Threads)
if(MSVC)
 target_compile_options(DeviceWorkflowTests PRIVATE /utf-8)
endif()
add_test(NAME DeviceWorkflowCore COMMAND DeviceWorkflowTests "${CMAKE_CURRENT_SOURCE_DIR}/../../..")
set_tests_properties(DeviceWorkflowCore PROPERTIES TIMEOUT 45 LABELS "functional;devices;polish")
