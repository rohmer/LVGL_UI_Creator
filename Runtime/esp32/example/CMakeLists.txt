cmake_minimum_required(VERSION 3.24)
get_filename_component(bundled_component "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)
if(EXISTS "${bundled_component}/lvgl-pb-component.json")
    set(default_component "${bundled_component}")
else()
    set(default_component "${CMAKE_CURRENT_LIST_DIR}/components/lvgl_pb_esp32")
endif()
set(LVGL_PB_COMPONENT_DIR "${default_component}" CACHE PATH "Generated standalone LVGL Protobuf component")
if(NOT EXISTS "${LVGL_PB_COMPONENT_DIR}/lvgl-pb-component.json")
    message(FATAL_ERROR "Generate a component first and set -DLVGL_PB_COMPONENT_DIR=absolute/path/to/component")
endif()
get_filename_component(example_component_name "${LVGL_PB_COMPONENT_DIR}" NAME)
set(EXAMPLE_PB_COMPONENT_NAME "${example_component_name}" CACHE INTERNAL "Generated component's IDF name" FORCE)
# ESP-IDF discovers requirements in a separate early-expansion CMake process.
set(ENV{LVGL_PB_EXAMPLE_COMPONENT_NAME} "${example_component_name}")
set(EXTRA_COMPONENT_DIRS "${LVGL_PB_COMPONENT_DIR}")
set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/../sdkconfig.defaults;${CMAKE_CURRENT_LIST_DIR}/sdkconfig.defaults")
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(lvgl_protobuf_display)
