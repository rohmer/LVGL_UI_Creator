cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(LVGLProtobufLinux VERSION 0.9.4 LANGUAGES C CXX)
endif()
include(CTest)
include(GNUInstallDirs)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(LVGL_PB_WITH_SDL "Build the SDL3 desktop shared library and viewer" ON)
option(LVGL_PB_BUNDLED_SDL "Build the bundled pinned SDL3 instead of finding a system package" OFF)
set(LVGL_PB_PROFILE_COMPONENT_DIR "" CACHE PATH "Optional generated project-specific runtime component")
get_filename_component(pb_root "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)

if(LVGL_PB_WITH_SDL)
    if(NOT LVGL_PB_BUNDLED_SDL)
        find_package(SDL3 CONFIG QUIET)
    endif()
    if(NOT TARGET SDL3::SDL3)
        if(NOT EXISTS "${pb_root}/3rdParty/SDL3/CMakeLists.txt")
            message(FATAL_ERROR "SDL3 development package is unavailable; use the complete source toolkit or set SDL3_DIR")
        endif()
        set(SDL_SHARED ON CACHE BOOL "" FORCE)
        set(SDL_STATIC OFF CACHE BOOL "" FORCE)
        set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
        set(SDL_TESTS OFF CACHE BOOL "" FORCE)
        set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
        add_subdirectory("${pb_root}/3rdParty/SDL3" "${CMAKE_CURRENT_BINARY_DIR}/sdl3" EXCLUDE_FROM_ALL)
        install(TARGETS SDL3-shared LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
    endif()
endif()

set(pb_drivers "${pb_root}/UICreator/Drivers/SDL3")
set(pb_sdl_sources)
if(LVGL_PB_WITH_SDL)
    foreach(name window keyboard mouse mousewheel sw)
        list(APPEND pb_sdl_sources "${pb_drivers}/lv_sdl_${name}.c")
    endforeach()
endif()
if(LVGL_PB_PROFILE_COMPONENT_DIR)
    get_filename_component(LVGL_PB_PROFILE_COMPONENT_DIR "${LVGL_PB_PROFILE_COMPONENT_DIR}" ABSOLUTE)
    add_subdirectory("${LVGL_PB_PROFILE_COMPONENT_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/profile")
    set(pb_core lvgl_pb_esp32)
    if(LVGL_PB_WITH_SDL)
        target_compile_definitions(${pb_core} PUBLIC LVGL_PB_ENABLE_SDL=1)
        target_sources(${pb_core} PRIVATE ${pb_sdl_sources})
        target_link_libraries(${pb_core} PUBLIC SDL3::SDL3)
    endif()
else()
    file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/config")
    if(LVGL_PB_WITH_SDL)
        set(pb_sdl 1)
    else()
        set(pb_sdl 0)
    endif()
    file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/config/lv_conf.h" CONTENT
        "#include \"${pb_root}/3rdParty/LVGL/lv_conf_v9.h\"\n#undef LV_USE_SDL\n#define LV_USE_SDL ${pb_sdl}\n" @ONLY)
    file(GLOB_RECURSE pb_lvgl_sources CONFIGURE_DEPENDS "${pb_root}/3rdParty/LVGL/lvgl/src/*.c")
    list(FILTER pb_lvgl_sources EXCLUDE REGEX "/drivers/sdl/")
    add_library(lvgl STATIC ${pb_lvgl_sources} ${pb_sdl_sources})
    add_library(lvgl::lvgl ALIAS lvgl)
    target_include_directories(lvgl PUBLIC "${pb_root}/3rdParty/LVGL/lvgl" "${pb_root}/3rdParty/LVGL/lvgl/include")
    target_compile_definitions(lvgl PUBLIC LV_KCONFIG_IGNORE=1 LV_CONF_PATH="${CMAKE_CURRENT_BINARY_DIR}/config/lv_conf.h")
    if(LVGL_PB_WITH_SDL)
        target_link_libraries(lvgl PUBLIC SDL3::SDL3)
    endif()
    set(UICREATOR_USE_LVGL9 ON)
    set(LVGL_RUNTIME_TARGET "9.6" CACHE STRING "" FORCE)
    add_subdirectory("${pb_root}/LVSerial" "${CMAKE_CURRENT_BINARY_DIR}/lvserial")
    target_compile_definitions(LVSerial PUBLIC LVSERIAL_EMBEDDED_RUNTIME=1)
    get_target_property(pb_serial_sources LVSerial SOURCES)
    list(FILTER pb_serial_sources EXCLUDE REGEX "/(PNGToArray|lodepng)\\.cpp$")
    set_property(TARGET LVSerial PROPERTY SOURCES "${pb_serial_sources}")
    add_subdirectory("${pb_root}/Runtime" "${CMAKE_CURRENT_BINARY_DIR}/runtime")
    set(pb_core LVGLRuntime_9_6)
endif()

add_library(LVGLProtobufLinux SHARED Board.cpp)
add_library(LVGLProtobuf::Linux ALIAS LVGLProtobufLinux)
target_compile_features(LVGLProtobufLinux PRIVATE cxx_std_17)
target_include_directories(LVGLProtobufLinux PUBLIC "${CMAKE_CURRENT_LIST_DIR}/include" "${pb_root}/Runtime/windows/include")
target_link_libraries(LVGLProtobufLinux PRIVATE ${pb_core})
set_target_properties(LVGLProtobufLinux PROPERTIES OUTPUT_NAME LVGLProtobufDisplay VERSION 0.9.4 SOVERSION 0 INSTALL_RPATH "$ORIGIN")
if(LVGL_PB_WITH_SDL)
    target_sources(LVGLProtobufLinux PRIVATE Display.cpp)
    target_link_libraries(LVGLProtobufLinux PRIVATE SDL3::SDL3)
    add_executable(lvgl-pb-viewer Viewer.cpp)
    target_link_libraries(lvgl-pb-viewer PRIVATE LVGLProtobufLinux)
    set_target_properties(lvgl-pb-viewer PROPERTIES INSTALL_RPATH "$ORIGIN/../${CMAKE_INSTALL_LIBDIR}")
    install(TARGETS lvgl-pb-viewer RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
install(TARGETS LVGLProtobufLinux LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES include/lvgl_pb_board.h "${pb_root}/Runtime/windows/include/lvgl_pb_display.h" DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
if(LVGL_PB_PROFILE_COMPONENT_DIR)
    install(FILES "${LVGL_PB_PROFILE_COMPONENT_DIR}/features.json" DESTINATION "${CMAKE_INSTALL_DATADIR}/lvgl-protobuf")
    install(DIRECTORY "${LVGL_PB_PROFILE_COMPONENT_DIR}/projects/" DESTINATION "${CMAKE_INSTALL_DATADIR}/lvgl-protobuf/projects")
    install(DIRECTORY "${LVGL_PB_PROFILE_COMPONENT_DIR}/licenses/" DESTINATION "${CMAKE_INSTALL_DATADIR}/lvgl-protobuf/licenses")
else()
    foreach(pair "LICENSE;LVGL_UI_Creator_LICENSE" "3rdParty/LVGL/lvgl/LICENCE.txt;LVGL_LICENSE")
        list(GET pair 0 source)
        list(GET pair 1 destination)
        if(EXISTS "${pb_root}/${source}")
            install(FILES "${pb_root}/${source}" DESTINATION "${CMAKE_INSTALL_DATADIR}/lvgl-protobuf/licenses" RENAME "${destination}")
        endif()
    endforeach()
    foreach(pair "KeyboardFont.LICENSE.txt;KeyboardFont_LICENSE"
            "KeyboardHindiFont.LICENSE.txt;KeyboardHindiFont_LICENSE.txt"
            "KeyboardBengaliFont.LICENSE.txt;KeyboardBengaliFont_LICENSE.txt"
            "KeyboardCyrillicFont.LICENSE.txt;KeyboardCyrillicFont_LICENSE.txt"
            "KeyboardGreekFont.LICENSE.txt;KeyboardGreekFont_LICENSE.txt"
            "KeyboardUrduFont.LICENSE.txt;KeyboardUrduFont_LICENSE.txt"
            "models/EnglishWordList.LICENSE.md;EnglishWordList_LICENSE.md"
            "PROVENANCE.md;Keyboard_PROVENANCE.md")
        list(GET pair 0 source)
        list(GET pair 1 destination)
        install(FILES "${ADVKBD_SOURCE_ROOT}/src/${source}" DESTINATION "${CMAKE_INSTALL_DATADIR}/lvgl-protobuf/licenses" RENAME "${destination}")
    endforeach()
    foreach(notice LICENSE NOTICE.md)
        install(FILES "${ADVKBD_SOURCE_ROOT}/${notice}" DESTINATION "${CMAKE_INSTALL_DATADIR}/lvgl-protobuf/licenses"
            RENAME "AdvancedKeyboard_${notice}")
    endforeach()
endif()
if(EXISTS "${pb_root}/3rdParty/SDL3/LICENSE.txt")
    install(FILES "${pb_root}/3rdParty/SDL3/LICENSE.txt" DESTINATION "${CMAKE_INSTALL_DATADIR}/lvgl-protobuf/licenses" RENAME SDL3_LICENSE)
endif()

if(BUILD_TESTING)
    if(NOT LVGL_PB_TEST_KEYBOARD_PROJECT AND EXISTS "${ADVKBD_SOURCE_ROOT}/src/models/enUS16.ngtr")
        find_package(Python3 COMPONENTS Interpreter QUIET)
        if(Python3_Interpreter_FOUND)
            set(LVGL_PB_TEST_KEYBOARD_PROJECT "${CMAKE_CURRENT_BINARY_DIR}/keyboard-trie.pb")
            execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/tests/make_fixture.py"
                "${pb_root}" "${LVGL_PB_TEST_KEYBOARD_PROJECT}" RESULT_VARIABLE pb_fixture_result)
            if(NOT pb_fixture_result EQUAL 0)
                message(FATAL_ERROR "Cannot create the Linux keyboard/trie fixture")
            endif()
        endif()
    endif()
    if(NOT LVGL_PB_TEST_PROJECT)
        set(LVGL_PB_TEST_PROJECT "${LVGL_PB_TEST_KEYBOARD_PROJECT}")
    endif()
    add_executable(LVGLProtobufLinuxTests tests/Board.c)
    target_link_libraries(LVGLProtobufLinuxTests PRIVATE LVGLProtobufLinux)
    add_test(NAME LVGLProtobufLinuxBoard COMMAND LVGLProtobufLinuxTests)
    if(LVGL_PB_TEST_KEYBOARD_PROJECT)
        add_executable(LVGLProtobufLinuxKeyboardTests tests/Keyboard.c)
        target_link_libraries(LVGLProtobufLinuxKeyboardTests PRIVATE LVGLProtobufLinux)
        add_test(NAME LVGLProtobufLinuxKeyboard COMMAND LVGLProtobufLinuxKeyboardTests "${LVGL_PB_TEST_KEYBOARD_PROJECT}")
        set_tests_properties(LVGLProtobufLinuxKeyboard PROPERTIES TIMEOUT 30)
    endif()
    if(LVGL_PB_WITH_SDL AND LVGL_PB_TEST_PROJECT)
        add_executable(LVGLProtobufLinuxWindowTests tests/Window.c)
        target_link_libraries(LVGLProtobufLinuxWindowTests PRIVATE LVGLProtobufLinux)
        add_test(NAME LVGLProtobufLinuxWindow COMMAND LVGLProtobufLinuxWindowTests "${LVGL_PB_TEST_PROJECT}" "${CMAKE_CURRENT_BINARY_DIR}/linux-screen.bmp")
        add_test(NAME LVGLProtobufLinuxViewer COMMAND lvgl-pb-viewer "${LVGL_PB_TEST_PROJECT}" --frames 4 --screenshot "${CMAKE_CURRENT_BINARY_DIR}/viewer-screen.bmp")
        set_tests_properties(LVGLProtobufLinuxWindow LVGLProtobufLinuxViewer PROPERTIES ENVIRONMENT "SDL_VIDEO_DRIVER=dummy;SDL_RENDER_DRIVER=software" TIMEOUT 30)
    endif()
endif()
