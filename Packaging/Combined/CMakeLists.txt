cmake_minimum_required(VERSION 3.24)
project(LVGLCombinedInstaller VERSION 0.9.0 LANGUAGES NONE)

if(NOT WIN32)
    message(FATAL_ERROR "The combined Windows installer requires Windows and NSIS.")
endif()
if(NOT DEFINED STAGE_ROOT OR NOT DEFINED RELEASE_VERSION)
    message(FATAL_ERROR "Use Packaging/BuildInstaller.ps1 with verified x64 and x86 payloads.")
endif()
get_filename_component(packaging "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)
get_filename_component(repository "${packaging}/.." ABSOLUTE)
file(READ "${STAGE_ROOT}/x64/architecture-manifest.json" manifest64)
file(READ "${STAGE_ROOT}/x86/architecture-manifest.json" manifest86)

# Identical assets, documentation and device SDKs are installed once. The
# architecture components overlap intentionally; exactly one can be selected.
foreach(architecture x64 x86)
    if(architecture STREQUAL "x64")
        set(manifest "${manifest64}")
    else()
        set(manifest "${manifest86}")
    endif()
    string(JSON count LENGTH "${manifest}" files)
    math(EXPR last "${count} - 1")
    foreach(index RANGE 0 ${last})
        string(JSON relative MEMBER "${manifest}" files ${index})
        string(JSON hash GET "${manifest}" files "${relative}")
        string(JSON other_hash ERROR_VARIABLE missing GET "${manifest64}" files "${relative}")
        if(architecture STREQUAL "x64")
            string(JSON other_hash ERROR_VARIABLE missing GET "${manifest86}" files "${relative}")
        endif()
        set(component "${architecture}")
        if(NOT missing AND hash STREQUAL other_hash)
            if(architecture STREQUAL "x86")
                continue()
            endif()
            set(component common)
        endif()
        get_filename_component(destination "${relative}" DIRECTORY)
        if(destination STREQUAL "")
            set(destination .)
        endif()
        install(FILES "${STAGE_ROOT}/${architecture}/${relative}" DESTINATION "${destination}" COMPONENT "${component}")
    endforeach()
    install(FILES "${STAGE_ROOT}/${architecture}/architecture-manifest.json" DESTINATION . COMPONENT "${architecture}")
endforeach()

set(CMAKE_MODULE_PATH "${CMAKE_CURRENT_LIST_DIR}/Templates")
set(CPACK_PACKAGE_NAME "LVGL-UI-Creator")
set(CPACK_PACKAGE_VENDOR "LVGL UI Creator contributors")

set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "LVGL UI Creator for Windows x64 and x86")
set(CPACK_PACKAGE_FILE_NAME "LVGL-UI-Creator-${RELEASE_VERSION}-windows-setup")
set(CPACK_PACKAGE_INSTALL_DIRECTORY "LVGL UI Creator")
set(CPACK_PACKAGE_INSTALL_REGISTRY_KEY "LVGL UI Creator")
set(CPACK_RESOURCE_FILE_LICENSE "${repository}/LICENSE")
set(CPACK_GENERATOR NSIS)
set(CPACK_SYSTEM_NAME windows)
set(CPACK_NSIS_COMPONENT_INSTALL ON)
set(CPACK_COMPONENTS_ALL common x64 x86)
set(CPACK_COMPONENT_COMMON_REQUIRED ON)
set(CPACK_COMPONENT_COMMON_HIDDEN ON)
set(CPACK_COMPONENT_X64_DISPLAY_NAME "Windows x64 (64-bit)")
set(CPACK_COMPONENT_X86_DISPLAY_NAME "Windows x86 (32-bit)")
set(CPACK_COMPONENT_X86_DISABLED ON)
set(CPACK_NSIS_PACKAGE_NAME "LVGL UI Creator")
set(CPACK_NSIS_DISPLAY_NAME "LVGL UI Creator ${RELEASE_VERSION}")
set(CPACK_NSIS_UNINSTALL_NAME Uninstall)
set(CPACK_NSIS_MUI_ICON "${packaging}/UICreator.ico")

set(CPACK_NSIS_MANIFEST_DPI_AWARE ON)

set(CPACK_NSIS_ENABLE_UNINSTALL_BEFORE_INSTALL OFF)
