cmake_minimum_required(VERSION 3.24)
project(AdvancedKeyboardCorpusTools NONE)
if(NOT TOOLS_ARCHITECTURE MATCHES "^(x64|x86)$" OR NOT RELEASE_VERSION MATCHES "^[0-9]+\.[0-9]+\.[0-9]+$")
    message(FATAL_ERROR "Select a Windows tools architecture and release version")
endif()
if(NOT EXISTS "${TOOLS_PAYLOAD}/corpus-tools-manifest.json")
    message(FATAL_ERROR "Run Packaging/BuildCorpusTools.ps1 to publish and verify the tools payload first")
endif()
install(DIRECTORY "${TOOLS_PAYLOAD}/" DESTINATION .)
set(CPACK_PACKAGE_NAME "AdvancedKeyboard-CorpusTools")
set(CPACK_PACKAGE_VERSION "${RELEASE_VERSION}")
set(CPACK_PACKAGE_FILE_NAME "AdvancedKeyboard-CorpusTools-${RELEASE_VERSION}-windows-${TOOLS_ARCHITECTURE}")
set(CPACK_GENERATOR ZIP)
include(CPack)
