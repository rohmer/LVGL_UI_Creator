cmake_minimum_required(VERSION 3.24)
project(LVGLSingleArchitectureInstaller LANGUAGES NONE)
if(NOT WIN32 OR NOT APPLICATION_ARCHITECTURE MATCHES "^(x64|x86)$")
    message(FATAL_ERROR "Select a Windows x64 or x86 application payload")
endif()
if(NOT DEFINED STAGE_ROOT OR NOT DEFINED RELEASE_VERSION)
    message(FATAL_ERROR "Use Packaging/BuildInstaller.ps1 with a verified payload")
endif()
get_filename_component(packaging "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)
get_filename_component(repository "${packaging}/.." ABSOLUTE)
set(payload "${STAGE_ROOT}/${APPLICATION_ARCHITECTURE}")
file(READ "${payload}/architecture-manifest.json" manifest)
string(JSON count LENGTH "${manifest}" files)
if(count LESS 1)
    message(FATAL_ERROR "The application payload is empty")
endif()
math(EXPR last "${count}-1")
foreach(index RANGE 0 ${last})
    string(JSON relative MEMBER "${manifest}" files ${index})
    get_filename_component(destination "${relative}" DIRECTORY)
    if(destination STREQUAL "")
        set(destination .)
    endif()
    install(FILES "${payload}/${relative}" DESTINATION "${destination}" COMPONENT application)
endforeach()
install(FILES "${payload}/architecture-manifest.json" DESTINATION . COMPONENT application)
set(CMAKE_MODULE_PATH "${CMAKE_CURRENT_LIST_DIR}/Templates")
set(CPACK_PACKAGE_NAME "LVGL-UI-Creator")
set(CPACK_PACKAGE_VENDOR "LVGL UI Creator contributors")
set(CPACK_PACKAGE_VERSION "${RELEASE_VERSION}")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "LVGL UI Creator for Windows ${APPLICATION_ARCHITECTURE}")
set(CPACK_PACKAGE_FILE_NAME "LVGL-UI-Creator-${RELEASE_VERSION}-windows-${APPLICATION_ARCHITECTURE}-setup")
set(CPACK_PACKAGE_INSTALL_DIRECTORY "LVGL UI Creator ${APPLICATION_ARCHITECTURE}")
set(CPACK_RESOURCE_FILE_LICENSE "${repository}/LICENSE")
set(CPACK_GENERATOR NSIS)
set(CPACK_SYSTEM_NAME "windows-${APPLICATION_ARCHITECTURE}")
set(CPACK_NSIS_COMPONENT_INSTALL ON)
set(CPACK_COMPONENTS_ALL application)
set(CPACK_COMPONENT_APPLICATION_REQUIRED ON)
set(CPACK_COMPONENT_APPLICATION_HIDDEN ON)
set(CPACK_NSIS_PACKAGE_NAME "LVGL UI Creator ${APPLICATION_ARCHITECTURE}")
set(CPACK_NSIS_DISPLAY_NAME "LVGL UI Creator ${RELEASE_VERSION} (${APPLICATION_ARCHITECTURE})")
set(CPACK_UICREATOR_ARCHITECTURE "${APPLICATION_ARCHITECTURE}")
set(CPACK_NSIS_UNINSTALL_NAME Uninstall)
set(CPACK_NSIS_MUI_ICON "${packaging}/UICreator.ico")
set(CPACK_NSIS_MUI_UNIICON "${packaging}/UICreator.ico")
set(CPACK_NSIS_MANIFEST_DPI_AWARE ON)
set(CPACK_NSIS_COMPRESSOR "/SOLID lzma")
set(CPACK_NSIS_MODIFY_PATH OFF)
include(CPack)
